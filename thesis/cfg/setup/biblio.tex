  \printnames[author]{author}%
}

\defbibenvironment{bibliography}
  {\list
     {\printtext[labelnumberwidth]{%
        \printfield{labelnumber}}}
     {\setlength{\labelwidth}{1.25cm}% ширина метки (например, "[1]")
      \setlength{\labelsep}{0.5em}% расстояние между номером и текстом
      \setlength{\leftmargin}{1.25cm}% отступ всей записи
      \setlength{\itemindent}{0pt}%
      \setlength{\itemsep}{0pt}% расстояние между записями
      \setlength{\parsep}{0pt}%
      \setlength{\topsep}{0pt}%
      \setlength{\partopsep}{0pt}%
      \setlength{\listparindent}{1.25cm}% отступ для новых абзацев внутри записи
      \setlength{\rightmargin}{0pt}%
      \setlength{\parskip}{0pt}%
      \setlength{\parindent}{1.25cm}%
     }}
  {\endlist}
  {\item}
